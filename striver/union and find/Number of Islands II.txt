#include <bits/stdc++.h>
using namespace std;

int find_parent(int x,vector<int>& parent){
    
    if(parent[x]==-1){
        return x;
    }else{
        return parent[x]=find_parent(parent[x],parent);
    }
}


bool unions(int x,int y,vector<int>& parent,vector<int>& rank){
    
    int x_rep=find_parent(x,parent);
    int y_rep=find_parent(y,parent);
    
    if(x_rep==y_rep){
        return false;
    }else{
        if(rank[x_rep]>rank[y_rep]){
            parent[y_rep]=x_rep;
        }else if(rank[y_rep]>rank[x_rep]){
            parent[x_rep]=y_rep;
        }else{
            parent[y_rep]=x_rep;
            rank[x_rep]++;
        }
        return true;
    }
    
    
}


int main() {
	// your code goes here
	int m=3;
	int n=3;
	vector<vector<int>> grid(m,vector<int>(n,0));
	
	vector<pair<int,int>> pos={{0, 0}, {0, 2}, {2, 0}, {2, 2}, {1, 1}, {0, 1}, {1, 0}, {1, 2}, {2, 1}};
	
	
	
	vector<int> parent(m*n,-1);
	vector<int> rank(m*n,0);
	
	vector<int> res;
	
	int totalisland=0;
	
	for(auto ele: pos){
	    int x=ele.first;
	    int y=ele.second;
	    
	    if(grid[x][y]){
	        res.push_back(totalisland);
	        continue;
	    }
	    
	    grid[x][y]=1;
	    totalisland++;
	    
	    //upward;
	    if(x>0){
	        if(grid[x-1][y]){
	            
	            int oldpos=x*n+y;
	            int newpos=(x-1)*n+y;
	            if(unions(oldpos,newpos,parent,rank)){
	                totalisland--;
	            }
	        }
	    }
	    
	    //downward;
	    if(x<m-1){
	        if(grid[x+1][y]){
	            
	            int oldpos=x*n+y;
	            int newpos=(x+1)*n+y;
	            
	            if(unions(oldpos,newpos,parent,rank)){
	                totalisland--;
	            }
	        }
	    }
	    
	    //right;
	    if(y<n-1){
	        if(grid[x][y+1]){
	            
	            int oldpos=x*n+y;
	            int newpos=x*n+(y+1);
	            if(unions(oldpos,newpos,parent,rank)){
	                totalisland--;
	            }
	        }
	    }
	    
	    //left;
	    if(y>0){
	        if(grid[x][y-1]){
	            
	            int oldpos=x*n+y;
	            int newpos=x*n+(y-1);
	            if(unions(oldpos,newpos,parent,rank)){
	                totalisland--;
	            }
	        }
	    }
	    
	    res.push_back(totalisland);
	    
	    
	}
	
	for(auto ele: res){
	    cout<<ele<<"   ";
	}
	
	
}
