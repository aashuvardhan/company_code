Explaination: By running the search query on

#include <bits/stdc++.h>
using namespace std;


struct TrieNode{
    unordered_map<char,TrieNode*> child;
    bool isEnd;
    
    TrieNode(){
        isEnd=false;
    }
};


void insert(TrieNode* root,string str){
    
    TrieNode* curr=root;
    
    for(auto ele: str){
        if(curr->child.find(ele)==curr->child.end()){
            curr->child[ele]=new TrieNode();
        }
        curr=curr->child[ele];
    }
    curr->isEnd=true;
    
    
}



void dfs_util(TrieNode* curr,string prefix,vector<string>& temp){
    
    if(curr->isEnd==true){
        temp.push_back(prefix);
    }
    
    for(char i='a';i<='z';i++){
        if(curr->child.find(i)!=curr->child.end()){
            
            dfs_util(curr->child[i],prefix+i,temp);
        }
    }
    
}


int main() {
	// your code goes here
	vector<string> arr={"geeikistest", "geeksforgeeks", "geeksfortest"};
	int n=arr.size();
	
	string s="geeips";
	
	TrieNode* root=new TrieNode();
	
	for(auto ele: arr){
	    insert(root,ele);
	}
	
	vector<vector<string>> res;
	
	TrieNode* prevNode = root;
	
	string prefix="";
	
	for(auto ele: s){
	    
	    vector<string> temp;
	    prefix+=ele;
	    
	    
	    if(prevNode==NULL or prevNode->child.find(ele)==prevNode->child.end()){
	        prevNode=NULL;
	        temp.push_back("0");
	        res.push_back(temp);
	    }else{
	        
	        prevNode=prevNode->child[ele];
	        
	        dfs_util(prevNode,prefix,temp);
	        res.push_back(temp);
	        
	    }
	}
	
	for(auto ele: res){
	    for(auto ite: ele){
	        cout<<ite<<"   ";
	    }
	    cout<<endl;
	}
	
	

}
